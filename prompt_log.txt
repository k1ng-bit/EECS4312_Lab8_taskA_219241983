Pseudocode
function suggest_slots(day, working_hours, busy_intervals, duration, n, buffer=0, candidate_window=None):

    # 1. Validate input
    if working_hours.start >= working_hours.end:
        raise ValueError

    if duration <= 0:
        raise ValueError

    if n < 0:
        raise ValueError

    if buffer < 0:
        raise ValueError

    if candidate_window exists and candidate_window.start >= candidate_window.end:
        raise ValueError

    for each busy interval:
        if interval.start >= interval.end:
            raise ValueError

    if n == 0:
        return []

    # 2. Convert times to datetime using the given day
    work_start = combine(day, working_hours.start)
    work_end = combine(day, working_hours.end)

    start_limit = work_start
    end_limit = work_end

    if candidate_window exists:
        candidate_start = combine(day, candidate_window.start)
        candidate_end = combine(day, candidate_window.end)

        start_limit = max(work_start, candidate_start)
        end_limit = min(work_end, candidate_end)

    if start_limit >= end_limit:
        return []

    latest_start = end_limit - duration
    if latest_start < start_limit:
        return []

    # 3. Expand busy intervals by buffer
    busy_ranges = empty list
    for each busy interval:
        busy_start = combine(day, interval.start) - buffer
        busy_end = combine(day, interval.end) + buffer
        add (busy_start, busy_end) to busy_ranges

    # 4. Sort and merge busy ranges
    sort busy_ranges by start time
    merged_busy = merge overlapping/adjacent ranges

    # 5. Scan for valid appointment starts
    result = empty list
    current = start_limit
    step = 1 minute   # or whatever granularity is required

    while current <= latest_start and len(result) < n:
        slot_end = current + duration
        conflict = false

        for each (busy_start, busy_end) in merged_busy:
            if current < busy_end AND busy_start < slot_end:
                conflict = true
                break

        if not conflict:
            add Slot(start_time=current.time()) to result

        current = current + step

    return result
Helper function pseudocode for merging
function merge_intervals(intervals):
    if intervals is empty:
        return []

    sort intervals
    merged = [first interval]

    for each remaining interval:
        last = merged[-1]

        if current.start <= last.end:
            merge them by extending the end to max(last.end, current.end)
        else:
            append current to merged

    return merged
